<script>
    function calculateTab1() {
        const sheets = parseFloat(document.getElementById('t1-sheets').value) || 0;
        const gsm = parseFloat(document.getElementById('t1-gsm').value) || 0;
        const dim1 = parseFloat(document.getElementById('t1-dim1').value) || 0;
        const dim2 = parseFloat(document.getElementById('t1-dim2').value) || 0;
        const rollWeight = parseFloat(document.getElementById('t1-roll').value) || 0;
        const palletCap = parseFloat(document.getElementById('t1-pallet').value) || 0;

        const netWeight = (sheets * gsm * dim1 * dim2) / 10000000;
        const wasteWeight = netWeight * 1.03;
        const rollsCount = rollWeight > 0 ? (netWeight / rollWeight) : 0;
        const palletsCount = palletCap > 0 ? Math.ceil(sheets / palletCap) : 0;

        document.getElementById('res1-net').innerText = netWeight.toFixed(2) + " كجم";
        document.getElementById('res1-waste').innerText = wasteWeight.toFixed(2) + " كجم";
        document.getElementById('res1-rolls').innerText = rollsCount.toFixed(2) + " بكرة";
        document.getElementById('res1-pallets').innerText = palletsCount + " بالتة";
    }

    function calculateTab2() {
        const roll = parseFloat(document.getElementById('t2-roll').value) || 0;
        const gsm = parseFloat(document.getElementById('t2-gsm').value) || 0;
        const dim1 = parseFloat(document.getElementById('t2-dim1').value) || 0;
        const dim2 = parseFloat(document.getElementById('t2-dim2').value) || 0;
        const actualSheets = parseFloat(document.getElementById('t2-actual').value) || 0;
        const reamSheets = parseFloat(document.getElementById('t2-ream').value) || 0;

        const divisor = (gsm * dim1 * dim2);
        if (divisor > 0) {
            const expected = Math.round((roll * 10000000) / divisor);
            const reamWeight = (reamSheets * gsm * dim1 * dim2) / 10000000;
            const reamsCount = reamSheets > 0 ? (actualSheets / reamSheets) : 0;
            const actualProducedWeight = (actualSheets * gsm * dim1 * dim2) / 10000000;
            const deficit = roll - actualProducedWeight;

            document.getElementById('res2-expected').innerText = expected + " فرخ";
            document.getElementById('res2-ream-weight').innerText = reamWeight.toFixed(2) + " كجم";
            document.getElementById('res2-reams-count').innerText = reamsCount.toFixed(2) + " رزمة";
            document.getElementById('res2-deficit').innerText = deficit.toFixed(2) + " ك";
        }
    }

    // إضافة مستمعين للأحداث لتحديث النتائج تلقائياً عند الكتابة
    document.querySelectorAll('#t1-sheets, #t1-gsm, #t1-dim1, #t1-dim2, #t1-roll, #t1-pallet').forEach(input => {
        input.addEventListener('input', calculateTab1);
    });

    document.querySelectorAll('#t2-roll, #t2-gsm, #t2-dim1, #t2-dim2, #t2-actual, #t2-ream').forEach(input => {
        input.addEventListener('input', calculateTab2);
    });

    // حساب مبدئي عند تحميل الصفحة
    calculateTab1();
    calculateTab2();
</script>
